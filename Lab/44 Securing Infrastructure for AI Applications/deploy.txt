== 1/8 Enable APIs
  $ gcloud services enable compute.googleapis.com notebooks.googleapis.com aiplatform.googleapis.com iam.googleapis.com storage.googleapis.com logging.googleapis.com --project=dynamic-circlet-j41fc --quiet

== 2/8 Private VPC and subnet
  $ gcloud compute networks create genai-secure-vpc --subnet-mode=custom --bgp-routing-mode=regional --project=dynamic-circlet-j41fc --quiet
  $ gcloud compute networks subnets create genai-subnet-us-central1 --network=genai-secure-vpc --region=us-central1 --range=10.0.1.0/24 --enable-private-ip-google-access --enable-flow-logs --logging-metadata=include-all --project=dynamic-circlet-j41fc --quiet

== 3/8 Default-deny firewall
  $ gcloud compute firewall-rules create genai-deny-all-ingress --network=genai-secure-vpc --direction=INGRESS --action=DENY --rules=all --source-ranges=0.0.0.0/0 --priority=65534 --enable-logging --logging-metadata=include-all --project=dynamic-circlet-j41fc --quiet

== 4/8 Outbound access (Cloud NAT)
  - skipped (pass --nat to allow pip/apt installs from the internet)

== 5/8 Least-privilege service account
  $ gcloud iam service-accounts create vertex-workbench-sa '--display-name=Vertex AI Workbench (least privilege)' --project=dynamic-circlet-j41fc --quiet
